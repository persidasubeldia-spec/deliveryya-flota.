// app-tienda.js - Módulo Frontend App Tienda DeliveryYa Flota class AppTiendaManager { constructor(supabaseClient, tiendaId) { this.supabase = supabaseClient; this.tiendaId = tiendaId; this.tarifasCache = {}; } async inicializarTarifas() { const { data, error } = await this.supabase.from('tarifas_delivery').select('*'); if (!error && data) { data.forEach(t => { this.tarifasCache[t.localidad] = t.precio_gs; }); } } calcularTarifa(localidad) { return this.tarifasCache[localidad] ?? null; } validarTelefonoParaguayo(tel) { const limpio = (tel || "").replace(/\D/g, ""); return /^09\d{8}$/.test(limpio); } async guardarPedido(formData) { // 1. Validaciones de Negocio if (!formData.cliente || !formData.cliente.trim()) { throw new Error("Ingresá el nombre del cliente."); } if (!this.validarTelefonoParaguayo(formData.telefono)) { throw new Error("Ingresá un celular paraguayo válido (formato 09XXXXXXXX)."); } if (!formData.direccion || !formData.direccion.trim()) { throw new Error("Ingresá la dirección de entrega."); } const precioMercaderia = Number(formData.precioMercaderia); if (!Number.isFinite(precioMercaderia) || precioMercaderia < 0) { throw new Error("Ingresá un precio de mercadería válido en Guaraníes."); } const envio = this.calcularTarifa(formData.localidad); if (envio === null) { throw new Error("La localidad seleccionada está fuera de cobertura o pendiente de tarifa."); } // 2. Cálculo estricto de cobro según modalidad let totalACobrar = 0; if (formData.modalidadPago === "Pendiente") { totalACobrar = precioMercaderia + envio; } else if (formData.modalidadPago === "Pagado a Tienda") { totalACobrar = envio; // Solo se cobra el delivery al recibir } else if (formData.modalidadPago === "Todo Pagado") { totalACobrar = 0; // Nada que cobrar en destino } // 3. Persistencia segura en Supabase con vinculación estricta de tienda const payloadPedido = { empresa_id: formData.empresaId, tienda_id: this.tiendaId, nombre_cliente_snapshot: formData.cliente.trim(), telefono_snapshot: formData.telefono.trim(), direccion_snapshot: formData.direccion.trim(), localidad: formData.localidad, latitude: formData.latitude || null, longitude: formData.longitude || null, precio_mercaderia_gs: precioMercaderia, precio_envio_gs: envio, modalidad_pago: formData.modalidadPago, total_a_cobrar_gs: totalACobrar, estado: "Pendiente", observaciones: formData.observaciones || "" }; const { data, error } = await this.supabase .from('pedidos') .insert([payloadPedido]) .select() .single(); if (error) { throw new Error("Error al registrar el pedido en el servidor: " + error.message); } return data; } async consultarPedidosTienda(filtroEstado = 'Todos') { let query = this.supabase .from('pedidos') .select('*') .eq('tienda_id', this.tiendaId) .order('created_at', { ascending: false }); if (filtroEstado !== 'Todos') { query = query.eq('estado', filtroEstado); } const { data, error } = await query; if (error) throw new Error("No se pudieron cargar los pedidos."); return data; } } -- Habilitar extensión UUID create extension if not exists "uuid-ossp"; -- 1. Empresas administradoras create table empresas ( id uuid primary key default uuid_generate_v4(), nombre text not null, estado text default 'activo', created_at timestamp with time zone default timezone('utc'::text, now()) not null ); -- 2. Tiendas vinculadas create table tiendas ( id uuid primary key default uuid_generate_v4(), empresa_id uuid references empresas(id) on delete cascade not null, nombre text not null, responsable text, telefono text, email_comercial text, direccion text, latitude decimal(10, 8), longitude decimal(11, 8), activa boolean default true, created_at timestamp with time zone default timezone('utc'::text, now()) not null ); -- 3. Miembros y Roles por Tienda create table miembros_tienda ( id uuid primary key default uuid_generate_v4(), tienda_id uuid references tiendas(id) on delete cascade not null, usuario_id uuid references auth.users(id) on delete cascade not null, rol text default 'operador', activo boolean default true ); -- 4. Tarifas de Delivery por Localidad (Guaraníes) create table tarifas_delivery ( id uuid primary key default uuid_generate_v4(), localidad text unique not null, precio_gs integer not null, activa boolean default true ); insert into tarifas_delivery (localidad, precio_gs) values ('Asunción', 25000), ('Central', 25000), ('Limpio', 35000), ('Itá', 35000), ('Itauguá', 35000), ('Ypacaraí', 35000), ('Nueva Italia', 45000), ('Villeta', 45000), ('Villa Hayes', 45000), ('San Pedro', 55000); -- 5. Clientes por Tienda (Agenda Comercial) create table clientes ( id uuid primary key default uuid_generate_v4(), tienda_id uuid references tiendas(id) on delete cascade not null, nombre text not null, telefono_normalizado text not null, created_at timestamp with time zone default timezone('utc'::text, now()) not null ); -- 6. Pedidos Operativos create table pedidos ( id uuid primary key default uuid_generate_v4(), empresa_id uuid references empresas(id) not null, tienda_id uuid references tiendas(id) not null, cliente_id uuid references clientes(id), nombre_cliente_snapshot text not null, telefono_snapshot text not null, direccion_snapshot text not null, localidad text not null, latitude decimal(10, 8), longitude decimal(11, 8), precio_mercaderia_gs integer not null default 0, precio_envio_gs integer not null, modalidad_pago text not null, -- 'Pendiente', 'Pagado a Tienda', 'Todo Pagado' total_a_cobrar_gs integer not null, estado text default 'Pendiente', observaciones text, created_at timestamp with time zone default timezone('utc'::text, now()) not null, updated_at timestamp with time zone default timezone('utc'::text, now()) not null ); -- HABILITAR ROW LEVEL SECURITY (RLS) alter table tiendas enable row level security; alter table miembros_tienda enable row level security; alter table clientes enable row level security; alter table pedidos enable row level security; -- Políticas de Seguridad: Cada tienda opera estrictamente sobre sus propios registros create policy "Acceso restringido a miembros de la tienda" on pedidos for all using ( tienda_id in ( select tienda_id from miembros_tienda where usuario_id = auth.uid() and activo = true ) ); create policy "Clientes visibles solo para su tienda" on clientes for all using ( tienda_id in ( select tienda_id from miembros_tienda where usuario_id = auth.uid() and activo = true ) ); DeliveryYa Flota - Plataforma Integral de Logística
